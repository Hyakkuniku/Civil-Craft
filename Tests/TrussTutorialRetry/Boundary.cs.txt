namespace UnityEngine.SceneManagement
{
    public struct Scene
    {
        public int handle; public string name; public bool isLoaded;
        public Scene(int handle, string name = "CanyonCrossing", bool loaded = true) { this.handle = handle; this.name = name; isLoaded = loaded; }
        public bool IsValid() => handle != 0;
        public static bool operator ==(Scene a, Scene b) => a.handle == b.handle;
        public static bool operator !=(Scene a, Scene b) => a.handle != b.handle;
        public override bool Equals(object value) => value is Scene other && other == this;
        public override int GetHashCode() => handle;
    }
}

    [AttributeUsage(AttributeTargets.Field)] public sealed class SerializeField : Attribute { }

    public class Object
    {
        public static readonly List<Object> All = new List<Object>();
        public bool Destroyed;
        public Object() { All.Add(this); }
        public static T FindObjectOfType<T>() where T : Object => All.OfType<T>().FirstOrDefault(value => !value.Destroyed);
        public static bool operator ==(Object a, Object b)
        {
            bool an = ReferenceEquals(a, null) || a.Destroyed, bn = ReferenceEquals(b, null) || b.Destroyed;
            return an || bn ? an && bn : ReferenceEquals(a, b);
        }
        public static bool operator !=(Object a, Object b) => !(a == b);
        public override bool Equals(object value) => ReferenceEquals(this, value);
        public override int GetHashCode() => System.Runtime.CompilerServices.RuntimeHelpers.GetHashCode(this);
    }
    public class GameObject : Object
    {
        public UnityEngine.SceneManagement.Scene scene = new UnityEngine.SceneManagement.Scene(1);
        public bool activeSelf = true;
        public bool activeInHierarchy => activeSelf;
        public void SetActive(bool value) { activeSelf = value; }
    }
    public class Component : Object { public GameObject gameObject = new GameObject(); }
    public class Coroutine { public IEnumerator routine; }
    public class MonoBehaviour : Component
    {
        public readonly List<Coroutine> scheduled = new List<Coroutine>();
        public Coroutine StartCoroutine(IEnumerator value)
        {
            var job = new Coroutine { routine = value }; scheduled.Add(job);
            value.MoveNext(); return job;
        }
        public void StopCoroutine(Coroutine value) { if (value != null) scheduled.Remove(value); }
    }
    public static class Resources { public static T[] FindObjectsOfTypeAll<T>() where T : Object => Object.All.OfType<T>().Where(value => !value.Destroyed).ToArray(); }
