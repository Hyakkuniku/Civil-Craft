namespace UnityEngine.SceneManagement
{
    public struct Scene : IEquatable<Scene>
    {
        public string name;
        public int handle;
        public bool isLoaded;
        public Scene(string name, int handle = 1, bool loaded = true) { this.name = name; this.handle = handle; isLoaded = loaded; }
        public bool IsValid() => handle > 0;
        public bool Equals(Scene other) => handle == other.handle;
        public override bool Equals(object value) => value is Scene scene && Equals(scene);
        public override int GetHashCode() => handle;
        public static bool operator ==(Scene left, Scene right) => left.Equals(right);
        public static bool operator !=(Scene left, Scene right) => !left.Equals(right);
    }
}

    public class Object
    {
        public bool Destroyed;
        public string name;
        public static bool operator ==(Object left, Object right)
        {
            bool leftNull = ReferenceEquals(left, null) || left.Destroyed;
            bool rightNull = ReferenceEquals(right, null) || right.Destroyed;
            return leftNull || rightNull ? leftNull && rightNull : ReferenceEquals(left, right);
        }
        public static bool operator !=(Object left, Object right) => !(left == right);
        public static implicit operator bool(Object value) => value != null;
        public override bool Equals(object value) => ReferenceEquals(this, value);
        public override int GetHashCode() => System.Runtime.CompilerServices.RuntimeHelpers.GetHashCode(this);
    }
    public class GameObject : Object
    {
        public UnityEngine.SceneManagement.Scene scene;
        public bool activeInHierarchy = true;
        public readonly Transform transform;
        private readonly List<Component> components = new();
        public GameObject(UnityEngine.SceneManagement.Scene scene) { this.scene = scene; transform = new Transform(this); }
        public T Attach<T>() where T : Component, new() { T value = new T { gameObject = this }; components.Add(value); return value; }
        public T GetComponent<T>() where T : Component => components.OfType<T>().FirstOrDefault(value => !value.Destroyed);
    }
    public class Component : Object
    {
        public GameObject gameObject;
        public Transform transform => gameObject.transform;
        public T GetComponent<T>() where T : Component => gameObject.GetComponent<T>();
        public bool TryGetComponent<T>(out T value) where T : Component { value = GetComponent<T>(); return value != null; }
    }
    public class Behaviour : Component
    {
        public bool enabled = true;
        public bool isActiveAndEnabled => !Destroyed && enabled && gameObject != null && gameObject.activeInHierarchy;
    }

    public class Transform
    {
        public readonly GameObject gameObject;
        public Vector3 position;
        public Vector3 localScale = Vector3.one;
        public float yaw;
        public Transform(GameObject gameObject) { this.gameObject = gameObject; }
        public Vector3 TransformPoint(Vector3 local)
        {
            Vector3 scaled = new Vector3(local.x * localScale.x, local.y * localScale.y, local.z * localScale.z);
            double radians = yaw * Math.PI / 180d;
            return position + new Vector3((float)(scaled.x * Math.Cos(radians) + scaled.z * Math.Sin(radians)), scaled.y,
                (float)(-scaled.x * Math.Sin(radians) + scaled.z * Math.Cos(radians)));
        }
        public Vector3 InverseTransformPoint(Vector3 world)
        {
            Vector3 value = world - position;
            double radians = -yaw * Math.PI / 180d;
            return new Vector3((float)(value.x * Math.Cos(radians) + value.z * Math.Sin(radians)) / localScale.x,
                value.y / localScale.y, (float)(-value.x * Math.Sin(radians) + value.z * Math.Cos(radians)) / localScale.z);
        }
    }
    public struct Vector3
    {
        public float x, y, z;
        public Vector3(float x, float y, float z) { this.x = x; this.y = y; this.z = z; }
        public static Vector3 zero => new(0f, 0f, 0f);
        public static Vector3 one => new(1f, 1f, 1f);
        public static Vector3 operator +(Vector3 left, Vector3 right) => new(left.x + right.x, left.y + right.y, left.z + right.z);
        public static Vector3 operator -(Vector3 left, Vector3 right) => new(left.x - right.x, left.y - right.y, left.z - right.z);
    }
