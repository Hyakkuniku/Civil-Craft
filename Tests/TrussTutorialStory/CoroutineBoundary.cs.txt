    public class MonoBehaviour
    {
        public bool isActiveAndEnabled = true;
        public GameObject gameObject = new GameObject();
        public List<Coroutine> routines = new List<Coroutine>();
        public int stopped;
        public T GetComponent<T>() where T : class => null;
        public static T FindObjectOfType<T>() where T : class => World.Dialogue as T;
        public Coroutine StartCoroutine(IEnumerator value)
        {
            var routine = new Coroutine(value); routines.Add(routine);
            routine.Step(); return routine;
        }
        public void StopCoroutine(Coroutine routine) { stopped++; routine.running = false; }
        public void Tick()
        {
            foreach (var routine in routines.ToArray()) if (routine.running) routine.Step();
        }
    }
    public sealed class Coroutine
    {
        public IEnumerator iterator;
        public bool running = true;
        public Coroutine(IEnumerator value) { iterator = value; }
        public void Step() { running = running && iterator.MoveNext(); }
    }
