using System;

namespace UnityEngine
{
    public class Object
    {
        public static readonly List<Object> all = new List<Object>();
        public static int searches;
        public string name;
        public Object() { all.Add(this); }
        public static T[] FindObjectsOfType<T>(bool includeInactive = false) where T : Object { searches++; return all.OfType<T>().Where(value => includeInactive || !(value is Behaviour b) || b.isActiveAndEnabled).ToArray(); }
        public static T FindObjectOfType<T>() where T : Object => FindObjectsOfType<T>().FirstOrDefault();
    }
    public class GameObject : Object
    {
        public GameObject(string value = "Scene Object") { name = value; transform = new Transform(this); }
        public Transform transform;
        public UnityEngine.SceneManagement.Scene scene = new UnityEngine.SceneManagement.Scene(1, "CanyonCrossing");
        public bool activeSelf = true;
        public bool activeInHierarchy => activeSelf;
        public void SetActive(bool value) { activeSelf = value; }
        public T GetComponent<T>() where T : class => all.OfType<T>().FirstOrDefault(value => value is Component c && c.gameObject == this);
        public T[] GetComponentsInChildren<T>(bool includeInactive) where T : Object => all.OfType<T>().Where(value => value is Component c && c.gameObject.scene == scene).ToArray();
    }
    public class Transform : Object { public GameObject gameObject; public Transform(GameObject owner) { gameObject = owner; } }
    public class Component : Object { public GameObject gameObject = new GameObject(); public Transform transform => gameObject.transform; }
    public class Behaviour : Component { public bool enabled = true; public bool isActiveAndEnabled => enabled && gameObject.activeInHierarchy; }
    public class MonoBehaviour : Behaviour { }
    public static class Resources { public static T[] FindObjectsOfTypeAll<T>() where T : Object => Object.all.OfType<T>().ToArray(); }
    public static class Time { public static float unscaledTime; public static float timeScale = 1f; }
    public static class Debug { public static void LogWarning(string value, Object context = null) { } }
}
namespace UnityEngine.SceneManagement
{
    public struct Scene
    {
        public int id; public string name; public bool isLoaded;
        public Scene(int id, string name) { this.id = id; this.name = name; isLoaded = true; }
        public bool IsValid() => id > 0;
        public GameObject[] GetRootGameObjects() { int sceneId = id; return UnityEngine.Object.all.OfType<GameObject>().Where(value => value.scene.id == sceneId).ToArray(); }
        public static bool operator ==(Scene a, Scene b) => a.id == b.id;
        public static bool operator !=(Scene a, Scene b) => a.id != b.id;
        public override bool Equals(object value) => value is Scene scene && scene == this;
        public override int GetHashCode() => id;
    }
}
