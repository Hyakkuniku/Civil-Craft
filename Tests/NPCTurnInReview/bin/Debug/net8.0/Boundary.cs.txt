public sealed class NPCProgressionManager
{
    public string progressionSaveId = "Mills";
    public bool persistProgression = true, IsTravelling, local = true, validationAvailable = true;
    public List<NPCProgressionPhase> phases = new List<NPCProgressionPhase>();
    public int currentPhaseIndex, optionalSequenceVersion, arrivalDialogueVersion, playedSteps, featureGrants, legacyMaterialCalls, legacyDialogueCalls;
    public bool isInvokingPhaseDialogueFinished;
    public NPCOrderedLearningFlow orderedLearningFlow;
    public HashSet<int> completedDialoguePhases = new HashSet<int>(), runningOptionalSequencePhases = new HashSet<int>(), introducedMaterialPhases = new HashSet<int>(), invokedInteractionPhases = new HashSet<int>();
    public NPCContractGiver contractGiver;
    public NPCProgressionPhase CurrentPhase => phases[currentPhaseIndex];
    private bool CanRunLocalProgression() => local;
    private bool HasPendingTravelAlmanacReview(int index, NPCProgressionPhase phase) => false;
    private bool TryHandleWaitingQuizInteraction(int index, NPCProgressionPhase phase) => runningOptionalSequencePhases.Contains(index);
    private bool TryHandleLessonReviewInteraction(int index, NPCProgressionPhase phase) => false;
    private bool NeedsRepeatableContinuationRetry(NPCProgressionPhase phase) => false;
    private bool TryValidateOrderedQuizzes(NPCProgressionPhase phase, out string error) { error = "Unavailable"; return validationAvailable; }
    private void TryStartPhaseDialogue(int index, NPCProgressionPhase phase) { legacyDialogueCalls++; }
    private void TryShowMaterialIntroduction(int index, NPCProgressionPhase phase) { legacyMaterialCalls++; }
    private void PlayOrderedOptionalStep(int index, NPCProgressionPhase phase, int step, bool showMaterial, bool displayedMaterial, int version) { playedSteps++; }
    private void GrantConfiguredPhaseFeature(NPCProgressionPhase phase) { featureGrants++; }
    private void CancelQuizWaitingReminder() { }
    private void CancelLessonReviewReminder() { }
    private void ClearTravelAlmanacReviewState() { }
    public void Offer() => HandleOfferDialogueCompleted(contractGiver);
    public void Tap() { contractGiver.suppressed = false; HandleNPCInteracted(contractGiver); }
    public void Start() => TryStartOrderedOptionalSequence(currentPhaseIndex, CurrentPhase);
    public void Finish(int version) => FinishOrderedOptionalSequence(currentPhaseIndex, CurrentPhase, false, version);
    public void Stop(int version) => StopOrderedOptionalSequence(currentPhaseIndex, version);
    public void Cancel() => CancelOrderedOptionalSequence();
    public bool CanStart() => CanStartOptionalSequenceForContractState(currentPhaseIndex, CurrentPhase);
    public bool NeedsTalk() => NeedsPhaseInteraction();
