public class AlmanacManager
{
    public static AlmanacManager Instance;
    public AlmanacQuizAccess access;
    public bool IsOpenAndReady = true;
    public bool IsOpeningOrClosing,blockClose,blockQuizSelection,deferOpenAnimation;
    public GameObject Panel=new(){activeSelf=true};
    public int opens, restores,selectAttempts,selectedCategory=-1;
    public Action close;
    public Action<int> OnCategoryChanged;
    public T GetComponent<T>() where T : class => access as T;
    public void CloseAlmanacThen(Action callback)
    {
        if (blockClose) return;
        IsOpenAndReady = false;
        IsOpeningOrClosing=true;
        close = callback;
    }
    public void FinishClose() { Action callback = close; close = null; restores++;IsOpeningOrClosing=false;Panel.SetActive(false); callback?.Invoke(); }
    public void OpenAlmanac() {opens++;if(deferOpenAnimation){IsOpeningOrClosing=true;IsOpenAndReady=false;}else FinishOpen();}
    public void FinishOpen(){IsOpeningOrClosing=false;IsOpenAndReady=true;Panel.SetActive(true);}
    public bool TrySelectQuizOfferCategory(int index){selectAttempts++;if(!IsOpenAndReady||IsOpeningOrClosing||blockQuizSelection||index!=4)return false;selectedCategory=index;OnCategoryChanged?.Invoke(index);return true;}
}
public static class QuizUIManager
{
    public static bool openSucceeds = true;
    public static int opens, cancellations;
    public static QuizData openedQuiz;
    public static QuizContext openedContext;
    public static UnityEngine.Object activeOwner;
    public static Action<QuizOutcome> closed;
    public static bool IsAnyQuizActive=>activeOwner!=null;
    public static bool TryOpen(QuizData quiz, QuizContext context, UnityEngine.Object owner, Action<QuizOutcome> callback)
    {
        opens++;
        if (!openSucceeds) return false;
        openedQuiz = quiz; openedContext = context; activeOwner = owner; closed = callback; return true;
    }
    public static void Resolve(QuizOutcome outcome) { Action<QuizOutcome> callback = closed; closed = null; activeOwner = null; callback?.Invoke(outcome); }
    public static void CancelOwnedSession(UnityEngine.Object owner)
    {
        if (activeOwner != owner) return;
        cancellations++;
        Resolve(QuizOutcome.Cancelled);
    }
}
