using System;
using System.Collections.Generic;

public class UnityObject
{
    public bool destroyed;
    public static bool operator ==(UnityObject a, UnityObject b) => ReferenceEquals(a, b) ||
        ((ReferenceEquals(a, null) || a.destroyed) && (ReferenceEquals(b, null) || b.destroyed));
    public static bool operator !=(UnityObject a, UnityObject b) => !(a == b);
    public override bool Equals(object other) => ReferenceEquals(this, other);
    public override int GetHashCode() => base.GetHashCode();
}
public struct Vector3
{
    public float x, y, z;
    public Vector3(float x, float y, float z) { this.x = x; this.y = y; this.z = z; }
    public static float Distance(Vector3 a, Vector3 b) => MathF.Sqrt((a.x-b.x)*(a.x-b.x)+(a.y-b.y)*(a.y-b.y)+(a.z-b.z)*(a.z-b.z));
}
public class GameObject : UnityObject
{
    public bool activeSelf = true;
    public Transform transform;
    public Dictionary<Type, UnityObject> components = new Dictionary<Type, UnityObject>();
    public bool activeInHierarchy => activeSelf && (transform.parent == null || transform.parent.gameObject.activeInHierarchy);
    public GameObject() { transform = new Transform(this); }
    public static GameObject FindGameObjectWithTag(string tag) => null;
}
public class Transform : UnityObject
{
    public GameObject gameObject;
    public Transform parent;
    public Vector3 position;
    public int parentLookups;
    public Transform(GameObject owner) { gameObject = owner; }
    public T GetComponentInParent<T>(bool includeInactive) where T : UnityObject
    {
        parentLookups++;
        for (Transform item = this; item != null; item = item.parent)
            if (item.gameObject.components.TryGetValue(typeof(T), out UnityObject value) && value != null) return (T)value;
        return null;
    }
}
