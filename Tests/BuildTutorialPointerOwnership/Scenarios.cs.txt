    static void PrepareRestore(Fixture fixture){fixture.Manager.inspectionHiddenSequence=fixture.Manager.currentSequence;fixture.Manager.inspectionHiddenStepIndex=fixture.Manager.currentStepIndex;}
    static object Call(object value,string method,params object[] args)=>value.GetType().GetMethod(method,BindingFlags.Instance|BindingFlags.NonPublic).Invoke(value,args);
    static void Check(bool value,string message){checks++;if(!value)throw new InvalidOperationException(message);}
