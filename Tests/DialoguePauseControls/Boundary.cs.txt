namespace UnityEngine

    public class GameObject
    {
        public int scene;
        public bool activeSelf = true;
        public bool activeInHierarchy => activeSelf;
        public readonly Dictionary<Type, object> components = new Dictionary<Type, object>();
        public void SetActive(bool value) { activeSelf = value; }
        public T GetComponent<T>() where T : class => components.TryGetValue(typeof(T), out object value) ? value as T : null;
    }
    public class MonoBehaviour
    {
        public static readonly List<MonoBehaviour> World = new List<MonoBehaviour>();
        public GameObject gameObject;
        public Transform transform;
        public bool enabled = true;
        public bool isActiveAndEnabled => enabled && gameObject.activeInHierarchy;
        public MonoBehaviour() { gameObject = new GameObject { scene = 1 }; transform = new Transform(gameObject); gameObject.components[GetType()] = this; World.Add(this); }
        public T GetComponent<T>() where T : class => gameObject.GetComponent<T>();
        public static T[] FindObjectsOfType<T>() where T : MonoBehaviour => World.OfType<T>().Where(value => value.gameObject.activeInHierarchy).ToArray();
        public static T FindObjectOfType<T>() where T : MonoBehaviour => FindObjectsOfType<T>().FirstOrDefault();
        public Coroutine StartCoroutine(IEnumerator enumerator) => new Coroutine();
        public void StopCoroutine(Coroutine coroutine) { }
    }
    public class Transform
    {
        public GameObject gameObject;
        public Transform(GameObject owner) { gameObject = owner; }
        public T GetComponent<T>() where T : class => gameObject.GetComponent<T>();
    }

    {
        public float x, y;
        public Vector2(float x, float y) { this.x = x; this.y = y; }
        public static Vector2 zero => new Vector2(0, 0);
        public float sqrMagnitude => x * x + y * y;
    }
