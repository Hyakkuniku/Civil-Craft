// LessonTrigger is an explicit neighboring-system boundary. Its production
// pending-query/persistence behavior is tested by the separate trigger suite.
public class LessonTrigger
{
    public static LessonTrigger Current;
    public NPCProgressionManager Owner;
    public PlayerDataManager SaveManager;
    public PlayerData SaveOwner;
    public NPCProgressionPhase Phase;
    public int PhaseIndex, resumes, lineRequests;
    public bool Pending, Active = true, ResumeResult = true;
    public Action OnResume;
    public string[] Lines = { "Good work inspecting the vehicle.", "Open your Almanac and review the Live Load lesson.", "Close the book when you are ready to continue." };
    public static bool TryGetPendingAlmanacReview(NPCProgressionManager npc, out LessonTrigger trigger)
    {
        trigger = Current;
        bool allowed = trigger != null && trigger.Active && trigger.Pending && ReferenceEquals(trigger.Owner, npc) &&
            trigger.SaveManager == PlayerDataManager.Instance && trigger.SaveOwner == PlayerDataManager.Instance?.CurrentData &&
            npc.isActiveAndEnabled && npc.gameObject.scene.IsValid() && npc.gameObject.scene.isLoaded &&
            npc.CurrentPhase == trigger.Phase && npc.currentPhaseIndex == trigger.PhaseIndex &&
            !(FusionConnectionManager.Instance != null && (FusionConnectionManager.Instance.IsGuestInHostWorld || FusionConnectionManager.Instance.IsGuestSaveProtected));
        if (!allowed) trigger = null;
        return allowed;
    }
    public bool TryResumeAlmanacReview() { resumes++; OnResume?.Invoke(); return ResumeResult; }
    public string[] GetAlmanacReviewReminder() { lineRequests++; return (string[])Lines.Clone(); }
}
public class DialogueManager : MonoBehaviour
{
    public static DialogueManager found;
    public static bool IsAnyDialogueVisible;
    public bool isDialogueActive, isTyping;
    public Action onDialogueEndCallback;
    public Coroutine typingCoroutine;
    public InputManager inputManager = new InputManager();
    public PlayerInteract playerInteract = new PlayerInteract();
    public Queue<string> sentences = new Queue<string>();
    public int starts, ends;
    public Dialogue last;
    public Animator speakerAnimator;
    public Transform speakerTransform;
    public void StartDialogue(Dialogue value, Action completed, Animator animator, Transform speaker)
    { starts++; last = value; onDialogueEndCallback = completed; speakerAnimator = animator; speakerTransform = speaker; isDialogueActive = IsAnyDialogueVisible = true; }
    public void Finish() { Action prior = onDialogueEndCallback; onDialogueEndCallback = null; isDialogueActive = IsAnyDialogueVisible = false; prior?.Invoke(); }
    public void EndDialogue() { ends++; Finish(); }
    // ACTUAL_CANCEL
}
