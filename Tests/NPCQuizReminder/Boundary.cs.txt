public class AlmanacQuizAccess : MonoBehaviour
{
    // ACTUAL_ACCESS
    public static int openRequests;
    public static UnityEngine.Object lastRequestedOwner;
    // Opening is the neighboring queue boundary: this suite executes the
    // actual reminder eligibility/callbacks, not the book's lifecycle queue.
    public static bool RequestOpenWaitingQuiz(UnityEngine.Object sessionOwner)
    {
        if (!TryGetWaitingQuiz(sessionOwner, out _, out _)) return false;
        openRequests++; lastRequestedOwner = sessionOwner; return true;
    }
    public static void CancelOwned(UnityEngine.Object sessionOwner) { var a = AlmanacManager.Instance?.access; if (a == null || a.owner != sessionOwner) return; a.closed = null; a.owner = null; }
    public void Offer(UnityEngine.Object value, QuizData data, QuizContext identity)
    { owner = value; quiz = data; context = identity; closed = _ => { }; saveManager = PlayerDataManager.Instance; saveOwner = saveManager.CurrentData; }
    public void SetStarting(bool value) { starting = value; }
}
public class DialogueManager : MonoBehaviour
{
    public static DialogueManager found;
    public static bool IsAnyDialogueVisible;
    public bool isDialogueActive, isTyping;
    public Action onDialogueEndCallback;
    public Coroutine typingCoroutine;
    public InputManager inputManager;
    public PlayerInteract playerInteract;
    public Queue<string> sentences = new Queue<string>();
    public int starts, ends;
    public Dialogue last;
    public Animator speakerAnimator;
    public Transform speakerTransform;
    public void StartDialogue(Dialogue value, Action completed, Animator animator, Transform speaker)
    { starts++; last = value; onDialogueEndCallback = completed; speakerAnimator = animator; speakerTransform = speaker; isDialogueActive = IsAnyDialogueVisible = true; }
    public void Finish() { Action c = onDialogueEndCallback; onDialogueEndCallback = null; isDialogueActive = IsAnyDialogueVisible = false; c?.Invoke(); }
    public void EndDialogue() { ends++; Finish(); }
    // ACTUAL_CANCEL
}
