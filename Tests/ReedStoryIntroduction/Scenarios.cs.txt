using System;
using System.Reflection;
using UnityEngine;
public static class Scenarios{
 static int checks;static void Check(bool okay,string name){checks++;if(!okay)throw new Exception(name);}static void Set(object target,string field,object value)=>target.GetType().GetField(field,BindingFlags.Instance|BindingFlags.NonPublic).SetValue(target,value);static void Call(object target,string method)=>target.GetType().GetMethod(method,BindingFlags.Instance|BindingFlags.NonPublic).Invoke(target,null);
 sealed class Visit{public ReedStoryIntroduction helper;public NPCProgressionManager npc;public NPCContractGiver giver;public CinematicDirector movie;public DialogueManager dialogue;public NPCOrderedLearningFlow learning;public Dialogue situation=new(),bending=new();}
 static Visit New(){Engine.Reset();Time.timeScale=1;DialogueManager.IsAnyDialogueActive=DialogueManager.IsAnyDialogueVisible=false;PlayerDataManager.Instance=new();FusionConnectionManager.Instance=new();LessonUIManager.Instance=new();PauseManager.Instance=new();UIPanelCoordinator.Instance=new();GameManager.Instance=new();QuizUIManager.IsAnyQuizActive=false;var v=new Visit();var go=new GameObject();v.helper=go.AddComponent<ReedStoryIntroduction>();v.npc=go.AddComponent<NPCProgressionManager>();v.giver=go.AddComponent<NPCContractGiver>();v.npc.giver=v.giver;v.learning=go.AddComponent<NPCOrderedLearningFlow>();v.movie=new GameObject().AddComponent<CinematicDirector>();v.dialogue=new GameObject().AddComponent<DialogueManager>();Set(v.helper,"npc",v.npc);Set(v.helper,"contractGiver",v.giver);Set(v.helper,"cinematic",v.movie);Set(v.helper,"dialogueManager",v.dialogue);Set(v.helper,"npcAnimator",go.AddComponent<Animator>());Set(v.helper,"bendingLesson",new LessonData());Set(v.helper,"situationDialogue",v.situation);Set(v.helper,"bendingDialogue",v.bending);return v;}
 static void Start(Visit v){v.helper.BeginIntroduction();Check(v.helper.IsRunning&&v.giver.locked,"Introduction owns one lock");Check(v.movie.plays==1&&v.movie.OnCinematicFinished.listeners.Count==1,"Own cinematic listener attached");}
 static void ToSituation(Visit v){Start(v);v.movie.Complete();Engine.Tick();Check(ReferenceEquals(v.dialogue.shown,v.situation),"Situation follows completed cinematic");Check(v.movie.OnCinematicFinished.listeners.Count==0,"Cinematic listener detached");}
 static void ToLesson(Visit v){ToSituation(v);v.dialogue.Finish();Engine.Tick();Check(v.dialogue.plays==1,"Wait for situation close animation");v.dialogue.CloseAnimation();Engine.Tick();Check(ReferenceEquals(v.dialogue.shown,v.bending),"Bending dialogue follows situation");v.dialogue.Finish();Engine.Tick();Check(v.learning.opened==0,"Wait for bending close animation");v.dialogue.CloseAnimation();Engine.Tick();Check(v.learning.opened==1&&v.npc.moves==0,"Lesson shown before contract phase");}
 static void Stopped(Visit v,string cause){Engine.Tick();Check(!v.helper.IsRunning&&v.npc.moves==0,cause+" cannot advance");Check(v.movie.OnCinematicFinished.listeners.Count==0,cause+" listener detached");}
 public static int Run(){
  var v=New();DialogueManager.IsAnyDialogueVisible=true;v.helper.BeginIntroduction();Check(v.movie.plays==0&&v.giver.locked,"Initial close animation waits under lock");v.helper.BeginIntroduction();Check(v.giver.changes==1,"Duplicate tap cannot restart or relock");DialogueManager.IsAnyDialogueVisible=false;Engine.Tick();Check(v.movie.plays==1,"Movie starts after prior close");
  v=New();ToLesson(v);v.npc.installNewLock=true;v.learning.Complete();Engine.Tick();Check(v.npc.moves==1&&v.npc.lastMove=="Reed Contract Give"&&!v.helper.IsRunning,"Genuine lesson close advances exactly once");Check(v.giver.locked&&v.giver.prompt=="New phase lock","Cleanup preserves next phase's lock");Call(v.helper,"OnDisable");Check(v.npc.moves==1&&v.giver.locked,"Later disable cannot erase next phase lock");
  foreach(string invalid in new[]{"guest","protected","profile","phase","travelling","unloaded","npcDisabled","helperDisabled"}){v=New();Start(v);switch(invalid){case "guest":FusionConnectionManager.Instance.IsGuestInHostWorld=true;break;case "protected":FusionConnectionManager.Instance.IsGuestSaveProtected=true;break;case "profile":PlayerDataManager.Instance.CurrentData=new();break;case "phase":v.npc.CurrentPhase=new(){phaseId="SideContract"};v.giver.SetProgressionInteractionLocked(true,"Next");break;case "travelling":v.npc.IsTravelling=true;break;case "unloaded":v.helper.gameObject.scene=new Scene{id=1,loaded=false};break;case "npcDisabled":v.npc.isActiveAndEnabled=false;break;case "helperDisabled":v.helper.isActiveAndEnabled=false;break;}Stopped(v,invalid);if(invalid=="phase")Check(v.giver.locked&&v.giver.prompt=="Next","Old flow never releases phase replacement lock");else Check(!v.giver.locked,"Same phase releases own lock on "+invalid);v.movie.Complete();Engine.Tick();Check(v.npc.moves==0&&v.dialogue.plays==0,"Late completion ignored after "+invalid);}
  v=New();Start(v);v.movie.Interrupt();Stopped(v,"Cinematic interruption");Check(!v.giver.locked,"Interrupted movie unlocks retry");v.helper.BeginIntroduction();Check(v.movie.plays==2,"Interrupted introduction can replay");
  v=New();v.movie.refuse=true;v.helper.BeginIntroduction();Check(!v.helper.IsRunning&&v.npc.moves==0&&!v.giver.locked,"Rejected movie never masquerades as completion");
  v=New();ToSituation(v);Call(v.helper,"OnDisable");Check(v.dialogue.cancelled==1&&!v.giver.locked,"Disable closes owned dialogue");Stopped(v,"Disabled dialogue owner");
  v=New();ToSituation(v);Action other=()=>{};v.dialogue.StartDialogue(new Dialogue(),other,null,null);Call(v.helper,"OnDisable");Check(ReferenceEquals(v.dialogue.callback,other)&&v.dialogue.cancelled==0,"Teardown preserves replacement dialogue");
  v=New();ToLesson(v);Call(v.helper,"OnDisable");Check(v.learning.cancelled==1&&!LessonUIManager.Instance.IsOpen&&!v.giver.locked,"Disable releases only owned pending lesson");Stopped(v,"Disabled lesson owner");
  v=New();ToSituation(v);v.dialogue.Finish();v.dialogue.CloseAnimation();Engine.Tick();v.dialogue.Finish();v.dialogue.CloseAnimation();v.learning.busy=true;Engine.Tick();Check(!v.helper.IsRunning&&v.learning.cancelled==0,"Rejected lesson does not cancel someone else's reader");
  v=New();ToLesson(v);v.learning.Fail();Stopped(v,"Unavailable lesson");Check(!v.giver.locked,"Failed lesson stays retryable");
  foreach(string blocker in new[]{"pause","panel","quiz","lesson","transition","mode","time"}){v=New();switch(blocker){case "pause":PauseManager.Instance.isPaused=true;break;case "panel":UIPanelCoordinator.Instance.HasOpenPanel=true;break;case "quiz":QuizUIManager.IsAnyQuizActive=true;break;case "lesson":LessonUIManager.Instance.IsOpen=true;break;case "transition":GameManager.Instance.IsTransitioning=true;break;case "mode":GameManager.Instance.CurrentState=GameManager.GameState.Other;break;case "time":Time.timeScale=0;break;}v.helper.BeginIntroduction();Check(v.helper.IsRunning&&v.movie.plays==0,"Wait for busy "+blocker);PlayerDataManager.Instance.CurrentData=new();Stopped(v,"Profile changes while waiting "+blocker);}
  foreach(string invalid in new[]{"guest","protected","phase","identity","nullProfile","scene","missingLesson","blankDialogue","wrongGiver"}){v=New();switch(invalid){case "guest":FusionConnectionManager.Instance.IsGuestInHostWorld=true;break;case "protected":FusionConnectionManager.Instance.IsGuestSaveProtected=true;break;case "phase":v.npc.CurrentPhase.phaseId="SideContract";break;case "identity":v.npc.ProgressionSaveId="NPC1";break;case "nullProfile":PlayerDataManager.Instance.CurrentData=null;break;case "scene":v.npc.gameObject.scene=new Scene{id=0,loaded=true};break;case "missingLesson":Set(v.helper,"bendingLesson",null);break;case "blankDialogue":v.bending.sentences=new[]{" "};break;case "wrongGiver":Set(v.helper,"contractGiver",new GameObject().AddComponent<NPCContractGiver>());break;}v.helper.BeginIntroduction();Check(!v.helper.IsRunning&&v.giver.changes==0&&v.movie.plays==0,"Invalid begin is read-only: "+invalid);}
  v=New();Start(v);Call(v.helper,"OnDestroy");Check(!v.helper.IsRunning&&!v.giver.locked&&v.movie.OnCinematicFinished.listeners.Count==0,"Destroy tears down owner");
  return checks;
 }
}
