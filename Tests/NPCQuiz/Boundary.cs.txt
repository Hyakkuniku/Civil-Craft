namespace UnityEditor
{
    public static class AssetDatabase
    {
        private static UnityEngine.Sprite inspected;
        public static bool TryGetGUIDAndLocalFileIdentifier(UnityEngine.Sprite sprite, out string guid, out long id)
        { inspected = sprite; guid = sprite.guid; id = sprite.localId; return !string.IsNullOrEmpty(guid); }
        public static string GetAssetPath(UnityEngine.Sprite sprite) { return sprite.guid; }
        public static string GetAssetDependencyHash(string path) { return inspected.hash; }
    }
}

public sealed class PlayerDataManager
{
    public PlayerData CurrentData = new PlayerData();
    public bool saveSucceeds = true;
    public int saves;
    public Action saveHook;
    public List<QuizProgressSaveData> committed;
    public bool TrySaveGame()
    {
        saves++;
        saveHook?.Invoke();
        if (saveSucceeds)
        {
            committed = new List<QuizProgressSaveData>();
            foreach (QuizProgressSaveData record in CurrentData.quizProgress) committed.Add(record.Clone());
        }
        return saveSucceeds;
    }
