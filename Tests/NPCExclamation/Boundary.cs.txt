    public struct Vector2
    {
        public float x, y;
        public Vector2(float x, float y) { this.x = x; this.y = y; }
        public float sqrMagnitude => x * x + y * y;
        public static Vector2 zero => new Vector2(0f, 0f);
        public static Vector2 down => new Vector2(0f, -1f);
        public static Vector2 operator +(Vector2 a, Vector2 b) => new Vector2(a.x + b.x, a.y + b.y);
        public static Vector2 operator *(Vector2 a, float scale) => new Vector2(a.x * scale, a.y * scale);
    }
    public struct Vector3
    {
        public float x, y, z;
        public Vector3(float x, float y, float z) { this.x = x; this.y = y; this.z = z; }
        public float sqrMagnitude => x * x + y * y + z * z;
        public static Vector3 operator -(Vector3 a, Vector3 b) => new Vector3(a.x - b.x, a.y - b.y, a.z - b.z);
        public static Vector3 zero => new Vector3(0f, 0f, 0f);
    }
    public static class Mathf
    {
        public static float Max(float a, float b) => Math.Max(a, b);
        public static float Abs(float value) => Math.Abs(value);
        public static float Min(float a, float b) => Math.Min(a, b);
        public const float PI = (float)Math.PI;
        public static float Cos(float angle) => (float)Math.Cos(angle);
        public static float Sin(float angle) => (float)Math.Sin(angle);
    }
    public class Transform
    {
        public Vector3 position;
        public Transform parent;
        public GameObject gameObject = new GameObject();
        private readonly Dictionary<Type, object> components = new Dictionary<Type, object>();
        public void AddComponent<T>(T component) where T : class => components[typeof(T)] = component;
        public bool TryGetComponent<T>(out T component) where T : class
        {
            bool found = components.TryGetValue(typeof(T), out object value);
            component = value as T;
            return found;
        }
        public T GetComponent<T>() where T : class { TryGetComponent(out T component); return component; }
        public bool IsChildOf(Transform ancestor)
        {
            for (Transform item = this; item != null; item = item.parent)
                if (item == ancestor) return true;
            return false;
        }
    }

    }

    public struct Rect
    {
        public float x, y, width, height;
        public Rect(float x, float y, float width, float height) { this.x = x; this.y = y; this.width = width; this.height = height; }
        public Vector2 center => new Vector2(x + width * .5f, y + height * .5f);
        public float xMin => x;
        public float yMin => y;
        public float xMax => x + width;
        public float yMax => y + height;
        public static Rect MinMaxRect(float xMin, float yMin, float xMax, float yMax) => new Rect(xMin, yMin, xMax - xMin, yMax - yMin);
        public bool Overlaps(Rect other) => other.xMax > xMin && other.xMin < xMax && other.yMax > yMin && other.yMin < yMax;
    }

public sealed class NPCProgressionManager
{
    public bool phaseHasArrived;
    public object movementRoutine;
    public bool isActiveAndEnabled = true;
    public GameObject gameObject = new GameObject();
    public List<NPCProgressionPhase> phases = new List<NPCProgressionPhase>();
    public int currentPhaseIndex;
    public Transform transform = new Transform();
    public readonly List<Renderer> RenderedChildren = new List<Renderer>();
    private readonly List<Renderer> markerPresenceRenderers = new List<Renderer>();
    public Vector3 authoredStartPosition;
    public float arrivalPadding = .05f, waypointArrivalDistance = .1f;
    public NPCProgressionMovementMode movementMode;
    public bool GroundProbeSucceeds;
    public float GroundProbeHeight;
    public int GroundProbeCount;
    // Saved dialogue completion and current owner-flow/giver availability are
    // integration boundaries; the shared policy and quiz-step classification
    // themselves are compiled unchanged from production.
    public NPCContractGiver contractGiver = new NPCContractGiver();
    public readonly HashSet<int> runningOptionalSequencePhases = new HashSet<int>();
    public readonly HashSet<int> savedCompletedDialogues = new HashSet<int>();
    public bool pendingTravelReview;
    private bool HasPendingTravelAlmanacReview(int index, NPCProgressionPhase phase) => pendingTravelReview;
    private bool HasCompletedPhaseDialogue(int index) => savedCompletedDialogues.Contains(index);
    public void UseInteractionGate(bool value) { onlyShowExclamationWhenInteractionNeeded = value; }
    public NPCProgressionManager() { transform = gameObject.transform; }
    private void GetComponentsInChildren(bool includeInactive, List<Renderer> result)
    {
        // Unity clears the supplied list before reporting the current subtree.
        result.Clear();
        result.AddRange(RenderedChildren);
    }
    private bool TryProjectWaypointToGround(Vector3 position, out Vector3 grounded)
    {
        GroundProbeCount++;
        grounded = new Vector3(position.x, GroundProbeHeight, position.z);
        return GroundProbeSucceeds;
    }

    // ACTUAL_NPC_POLICY

    public bool Reached(int index) => HasReachedPhaseDestination(index);
    public void Activate(int index)
    {
        currentPhaseIndex = index;
        phaseHasArrived = HasReachedPhaseDestination(index);
    }
    public void BeginTravel() { phaseHasArrived = false; movementRoutine = new object(); }
    public void FinishTravel(int index) { movementRoutine = null; Activate(index); }
}

public static class UIReservedRegionLayout
{
    private static readonly HashSet<RectTransform> reservedRects = new HashSet<RectTransform>();
    public static void NotifyLayoutChanging() { }
    // Unity's transform/camera corner projection is an engine boundary. The tests
    // inject current physical-pixel bounds; visibility, padding, and intersection
    // behavior are compiled directly from the production implementation.
    public static bool TryGetScreenRect(RectTransform rect, out Rect bounds)
    {
        bounds = rect.ScreenBounds;
        return rect.BoundsValid && bounds.width > .5f && bounds.height > .5f;
    }
    // ACTUAL_HUD_RESERVATIONS
    public static void Reset() => reservedRects.Clear();
}
