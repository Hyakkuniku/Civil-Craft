public static class Scenarios
{
    private static int assertions;
    public static int Run()
    {
        EligibilityMatrix();
        InteractionNeededPolicy();
        ContinuationRetryPolicy();
        ModelPresence();
        SpawnReadiness();
        DestinationResolution();
        ArrivalLifecycle();
        GraphicGeometry();
        MapProjectionAndSize();
        HUDReservationGeometry();
        HUDReservationVisibility();
        return assertions;
    }

    private static NPCProgressionManager Create()
    {
        FusionConnectionManager.Instance = null;
        NPCSpawnConditionMonitor.Reset();
        NPCProgressionManager manager = new NPCProgressionManager();
        manager.RenderedChildren.Add(Model());
        manager.phases.Add(new NPCProgressionPhase { showExclamationMark = true, targetLocation = new Transform { position = new Vector3(10f, 2f, 20f) } });
        return manager;
    }

    private static void EligibilityMatrix()
    {
        // Every eligibility bit is independently necessary. No active checkbox
        // can expose an in-transit, unloaded, disabled, or foreign-host NPC.
        for (int bits = 0; bits < 128; bits++)
        {
            NPCProgressionManager manager = Create();
            bool arrived = (bits & 1) != 0;
            bool travelling = (bits & 2) != 0;
            bool enabled = (bits & 4) != 0;
            bool valid = (bits & 8) != 0;
            bool loaded = (bits & 16) != 0;
            bool guest = (bits & 32) != 0;
            bool checkedPhase = (bits & 64) != 0;
            manager.phaseHasArrived = arrived;
            manager.movementRoutine = travelling ? new object() : null;
            manager.isActiveAndEnabled = enabled;
            manager.gameObject.scene.valid = valid;
            manager.gameObject.scene.isLoaded = loaded;
            manager.phases[0].showExclamationMark = checkedPhase;
            FusionConnectionManager.Instance = new FusionConnectionManager { IsGuestInHostWorld = guest };
            Expect(manager.ShouldShowExclamationMark == (arrived && !travelling && enabled && valid && loaded && !guest && checkedPhase), "Eligibility bits " + bits);
        }
        NPCProgressionManager invalid = Create();
        invalid.phaseHasArrived = true;
        invalid.currentPhaseIndex = -1;
        Expect(!invalid.ShouldShowExclamationMark, "No marker before a phase exists");
        invalid.currentPhaseIndex = invalid.phases.Count;
        Expect(!invalid.ShouldShowExclamationMark, "No marker after sequence end");
        invalid.currentPhaseIndex = 0;
        invalid.phases[0] = null;
        Expect(!invalid.ShouldShowExclamationMark, "No marker for missing phase");
    }

    private static void InteractionNeededPolicy()
    {
        NPCProgressionManager npc = Create();
        npc.phaseHasArrived = true;
        npc.UseInteractionGate(true);
        Expect(npc.ShouldShowExclamationMark, "Unfinished arrived conversation calls for attention.");
        npc.savedCompletedDialogues.Add(0);
        Expect(!npc.ShouldShowExclamationMark, "Finishing a contract-free conversation clears its attention, including restored completion.");
        npc.savedCompletedDialogues.Clear();
        npc.runningOptionalSequencePhases.Add(0);
        Expect(!npc.ShouldShowExclamationMark, "Owned ordered dialogue/material/lesson/quiz flow hides its marker while it awaits player input elsewhere.");
        npc.runningOptionalSequencePhases.Clear();
        Expect(npc.ShouldShowExclamationMark, "Cancelling an unfinished ordered flow restores the talk marker for retry.");
        npc.runningOptionalSequencePhases.Add(1);
        Expect(npc.ShouldShowExclamationMark, "Stale another-phase sequence state cannot hide the current conversation.");
        npc.runningOptionalSequencePhases.Clear();
        npc.pendingTravelReview = true;
        Expect(!npc.ShouldShowExclamationMark, "Arrival while the walking-time Almanac review is pending cannot advertise another talk.");
        npc.pendingTravelReview = false;
        LessonTrigger.pendingOwner = npc;
        Expect(!npc.ShouldShowExclamationMark, "Pending vehicle-inspection Almanac review keeps attention on the book.");
        LessonTrigger.pendingOwner = Create();
        Expect(npc.ShouldShowExclamationMark, "Another NPC's pending review does not suppress this talk.");
        LessonTrigger.pendingOwner = null;
        DialogueManager.IsAnyDialogueVisible = true;
        Expect(!npc.ShouldShowExclamationMark, "Visible dialogue suppresses attention until its closing presentation finishes.");
        DialogueManager.IsAnyDialogueVisible = false;
        QuizUIManager.IsAnyQuizActive = true;
        Expect(!npc.ShouldShowExclamationMark, "An active quiz cannot advertise an overlapping talk.");
        QuizUIManager.IsAnyQuizActive = false;
        PauseManager.Instance = new PauseManager { isPaused = true };
        Expect(!npc.ShouldShowExclamationMark, "Pause suppresses opt-in attention.");
        PauseManager.Instance = null;
        Time.timeScale = 0;
        Expect(!npc.ShouldShowExclamationMark, "An externally paused game cannot request a conversation.");
        Time.timeScale = 1;
        GameManager.Instance = new GameManager { CurrentState = GameManager.GameState.Building };
        Expect(!npc.ShouldShowExclamationMark, "Build mode keeps the marker hidden.");
        GameManager.Instance.CurrentState = GameManager.GameState.CargoTesting;
        Expect(!npc.ShouldShowExclamationMark, "Cargo simulation keeps the marker hidden.");
        GameManager.Instance.CurrentState = GameManager.GameState.Normal;
        GameManager.Instance.IsTransitioning = true;
        Expect(!npc.ShouldShowExclamationMark, "Interrupted transitions cannot flash talk attention.");
        GameManager.Instance.IsTransitioning = false;
        Expect(npc.ShouldShowExclamationMark, "Returning to ordinary play restores an unfinished conversation.");

        ContractSO contract = new ContractSO();
        npc.phases[0].contract = contract;
        npc.contractGiver.contractToGive = contract;
        Expect(npc.ShouldShowExclamationMark, "An available unaccepted contract needs a conversation.");
        npc.contractGiver.HasAcceptedProgressionContract = true;
        Expect(!npc.ShouldShowExclamationMark, "Accepted incomplete contract points the player toward building, not the giver.");
        npc.contractGiver.isContractCompleted = true;
        Expect(npc.ShouldShowExclamationMark, "A completed crossing advertises the turn-in conversation.");
        npc.contractGiver.CanRestoreContractGuide = false;
        Expect(!npc.ShouldShowExclamationMark, "Fully turned-in, locked, or pending-offer givers cannot request another turn-in.");
        npc.contractGiver.CanRestoreContractGuide = true;
        npc.contractGiver.isContractCompleted = false;
        npc.phases[0].optionalDialogueSequence.Add(new NPCOptionalDialogueStep { stepType = NPCOptionalDialogueStep.StepType.Quiz });
        Expect(npc.ShouldShowExclamationMark, "A cancelled accepted-contract quiz may be resumed by talking, including after reload.");
        npc.runningOptionalSequencePhases.Add(0);
        Expect(!npc.ShouldShowExclamationMark, "Once offered in the Almanac the same contract quiz no longer requests a second talk.");
        npc.runningOptionalSequencePhases.Clear();
        npc.savedCompletedDialogues.Add(0);
        Expect(!npc.ShouldShowExclamationMark, "Finishing the contract's quiz returns attention to the unfinished build task.");
        npc.contractGiver.isContractCompleted = true;
        Expect(npc.ShouldShowExclamationMark, "Quiz completion must not hide a later genuine contract turn-in.");
        npc.contractGiver.contractToGive = new ContractSO();
        Expect(!npc.ShouldShowExclamationMark, "A stale giver contract cannot advertise the current phase's turn-in.");
        npc.contractGiver.contractToGive = contract;
        npc.contractGiver.IsInteractionAvailable = false;
        Expect(!npc.ShouldShowExclamationMark, "Unavailable cargo-gated interaction cannot show talk attention.");
        npc.contractGiver.IsInteractionAvailable = true;
        npc.contractGiver.isActiveAndEnabled = false;
        Expect(!npc.ShouldShowExclamationMark, "Disabled interaction cannot advertise talk.");
        npc.contractGiver = null;
        Expect(!npc.ShouldShowExclamationMark, "A missing interaction component cannot advertise talk.");

        // Opting out preserves the original phase-only behavior, independently
        // of every new dependency; these settings cannot affect other NPCs.
        npc.UseInteractionGate(false);
        DialogueManager.IsAnyDialogueVisible = QuizUIManager.IsAnyQuizActive = true;
        GameManager.Instance.IsTransitioning = true;
        npc.runningOptionalSequencePhases.Add(0);
        Expect(npc.ShouldShowExclamationMark, "Existing phase-only NPCs remain eligible despite all new attention-policy blockers.");
        npc.RenderedChildren.Clear();
        Expect(!npc.ShouldShowExclamationMark, "Opt-out never bypasses actual spawned-model presence.");
        DialogueManager.IsAnyDialogueVisible = QuizUIManager.IsAnyQuizActive = false;
        GameManager.Instance = null;
    }

    private static void ContinuationRetryPolicy()
    {
        NPCProgressionManager npc = Create(); npc.phaseHasArrived = true; npc.UseInteractionGate(true);
        NPCProgressionPhase phase = npc.phases[0]; phase.repeatDialogue = true;
        npc.savedCompletedDialogues.Add(0);
        Expect(!npc.ShouldShowExclamationMark, "A completed ordinary repeat reminder, including Bhan's waiting dialogue, needs no fresh talk marker.");
        var movie = new CinematicDirector { gameObject = new GameObject() };
        movie.gameObject.scene = npc.gameObject.scene;
        phase.onDialogueFinished.calls.Add((movie, "PlayCinematic", UnityEngine.Events.UnityEventCallState.RuntimeOnly));
        Expect(npc.ShouldShowExclamationMark, "A repeatable camera introduction still in its completed dialogue phase advertises retry after interruption.");
        movie.IsPlaying = true;
        Expect(!npc.ShouldShowExclamationMark, "A running camera continuation retains attention on its presentation instead of advertising another conversation.");
        movie.IsPlaying = false;
        Expect(npc.ShouldShowExclamationMark, "Stopping the unfinished camera tour restores the talk marker without clearing saved dialogue completion.");
        movie.isActiveAndEnabled = false;
        Expect(!npc.ShouldShowExclamationMark, "A disabled or unavailable camera continuation cannot advertise a usable retry.");
        movie.isActiveAndEnabled = true; movie.gameObject.scene = new Scene();
        Expect(!npc.ShouldShowExclamationMark, "An unrelated scene's camera cannot make this NPC need another conversation.");
        movie.gameObject.scene = npc.gameObject.scene;
        phase.repeatDialogue = false;
        Expect(!npc.ShouldShowExclamationMark, "Nonrepeating completed conversations cannot request a replay they do not support.");
        phase.repeatDialogue = true;
        phase.onDialogueFinished.calls[0] = (movie, "PlayCinematic", UnityEngine.Events.UnityEventCallState.Off);
        Expect(!npc.ShouldShowExclamationMark, "Disabled persistent callbacks cannot make a completed phase look retryable.");
        phase.onDialogueFinished.calls[0] = (movie, "UnrelatedMethod", UnityEngine.Events.UnityEventCallState.RuntimeOnly);
        Expect(!npc.ShouldShowExclamationMark, "Only the actual camera-start callback can justify camera-continuation retry attention.");
        phase.onDialogueFinished.calls[0] = (movie, "PlayContractOfferCinematic", UnityEngine.Events.UnityEventCallState.RuntimeOnly);
        Expect(npc.ShouldShowExclamationMark, "The equivalent explicitly repeatable camera-start entry supports the same interrupted retry.");
        phase.onDialogueFinished.calls.Add((new object(), "StartTutorial", UnityEngine.Events.UnityEventCallState.RuntimeOnly));
        Expect(npc.ShouldShowExclamationMark, "Ordinary companion callbacks do not become marker owners or block a proven camera retry.");
        var busyMovie = new CinematicDirector { gameObject = movie.gameObject, IsPlaying = true };
        phase.onDialogueFinished.calls.Add((busyMovie, "PlayCinematic", UnityEngine.Events.UnityEventCallState.RuntimeOnly));
        Expect(!npc.ShouldShowExclamationMark, "Any still-active authored camera continuation prevents a premature retry marker.");
        phase.onDialogueFinished.calls.Clear();
        var introduction = new ReedStoryIntroduction { gameObject = npc.gameObject };
        phase.onDialogueFinished.calls.Add((introduction, "BeginIntroduction", UnityEngine.Events.UnityEventCallState.RuntimeOnly));
        Expect(npc.ShouldShowExclamationMark, "Reed can be prompted to retry his interrupted combined tour and lesson flow.");
        introduction.IsRunning = true;
        Expect(!npc.ShouldShowExclamationMark, "Reed's running introduction retains its lock while it waits between owned camera/dialogue/lesson presentations.");
        introduction.IsRunning = false; introduction.isActiveAndEnabled = false;
        Expect(!npc.ShouldShowExclamationMark, "An unavailable Reed helper cannot advertise an actionable conversation.");
        introduction.isActiveAndEnabled = true; introduction.gameObject = new GameObject();
        Expect(!npc.ShouldShowExclamationMark, "A helper on another NPC cannot authorize this NPC's replay marker.");
        introduction.gameObject = npc.gameObject;
        phase.onDialogueFinished.calls[0] = (introduction, "WrongMethod", UnityEngine.Events.UnityEventCallState.RuntimeOnly);
        Expect(!npc.ShouldShowExclamationMark, "Only Reed's actual introduction entry is eligible for this narrowly scoped retry policy.");
        phase.onDialogueFinished.calls[0] = (introduction, "BeginIntroduction", UnityEngine.Events.UnityEventCallState.RuntimeOnly);
        npc.runningOptionalSequencePhases.Add(0);
        Expect(!npc.ShouldShowExclamationMark, "Owned ordered learning flows continue to suppress retry markers.");
        npc.runningOptionalSequencePhases.Clear();
        DialogueManager.IsAnyDialogueVisible = true;
        Expect(!npc.ShouldShowExclamationMark, "Dialogue closing presentation still suppresses interrupted-continuation attention.");
        DialogueManager.IsAnyDialogueVisible = false;
        GameManager.Instance = new GameManager { CurrentState = GameManager.GameState.Building };
        Expect(!npc.ShouldShowExclamationMark, "Camera retry attention cannot leak into Build Mode.");
        GameManager.Instance = null;
        npc.contractGiver.IsInteractionAvailable = false;
        Expect(!npc.ShouldShowExclamationMark, "A gated giver remains unavailable even when its continuation can be replayed.");
        npc.contractGiver.IsInteractionAvailable = true;
        phase.contract = new ContractSO(); npc.contractGiver.contractToGive = phase.contract; npc.contractGiver.HasAcceptedProgressionContract = true;
        Expect(!npc.ShouldShowExclamationMark, "A cinematic callback never overrides accepted incomplete contract attention policy.");
        phase.contract = null;
        npc.RenderedChildren.Clear();
        Expect(!npc.ShouldShowExclamationMark, "An unspawned model cannot advertise an interrupted tour.");
        npc.RenderedChildren.Add(Model());
        phase.onDialogueFinished = null;
        Expect(!npc.ShouldShowExclamationMark, "Missing callbacks cannot manufacture attention on a completed repeat phase.");
        phase.onDialogueFinished = new UnityEngine.Events.UnityEvent();
        phase.onDialogueFinished.calls.Add((new object(), "StartTutorial", UnityEngine.Events.UnityEventCallState.RuntimeOnly));
        Expect(!npc.ShouldShowExclamationMark, "Completed tutorial callbacks and unrelated repeat reminders retain their old no-attention behavior.");
        npc.UseInteractionGate(false); movie.IsPlaying = true;
        phase.onDialogueFinished.calls[0] = (movie, "PlayCinematic", UnityEngine.Events.UnityEventCallState.RuntimeOnly);
        Expect(npc.ShouldShowExclamationMark, "NPCs that did not opt into interaction-needed attention retain their authored phase-only policy.");
        Expect(npc.savedCompletedDialogues.Contains(0), "Marker eligibility never clears or rewrites completed dialogue progression.");
    }
    private static void ModelPresence()
    {
        NPCProgressionManager npc = Create();
        npc.phaseHasArrived = true;
        Expect(npc.ShouldShowExclamationMark, "Spawned arrived model is eligible before either marker projects it");
        npc.RenderedChildren.Clear();
        Expect(!npc.ShouldShowExclamationMark, "An active progression holder without a spawned model cannot show any exclamation");
        npc.RenderedChildren.Add(null);
        Expect(!npc.ShouldShowExclamationMark, "A removed/destroyed model cannot leave a cached visible badge");
        npc.RenderedChildren.Add(new ParticleSystemRenderer());
        Expect(!npc.ShouldShowExclamationMark, "Particles or spawn effects cannot impersonate the NPC body");
        MeshRenderer nameLabel = new MeshRenderer();
        nameLabel.gameObject.transform.AddComponent(new TMPro.TMP_Text());
        npc.RenderedChildren.Add(nameLabel);
        Expect(!npc.ShouldShowExclamationMark, "A mesh-rendered name label does not count as a present NPC");
        npc.RenderedChildren.Add(new Renderer());
        Expect(!npc.ShouldShowExclamationMark, "An arbitrary non-model renderer cannot count as the NPC");
        npc.RenderedChildren.Clear();
        MeshRenderer emptyStatic = new MeshRenderer();
        npc.RenderedChildren.Add(emptyStatic);
        Expect(!npc.ShouldShowExclamationMark, "A placeholder MeshRenderer with no MeshFilter is not a present model");
        emptyStatic.gameObject.transform.AddComponent(new MeshFilter());
        Expect(!npc.ShouldShowExclamationMark, "A MeshFilter whose mesh is not loaded cannot show a premature marker");
        npc.RenderedChildren.Clear();
        SkinnedMeshRenderer emptySkinned = new SkinnedMeshRenderer();
        npc.RenderedChildren.Add(emptySkinned);
        Expect(!npc.ShouldShowExclamationMark, "A placeholder skinned model with no mesh cannot show a premature marker");
        emptySkinned.sharedMesh = new Mesh();
        Expect(npc.ShouldShowExclamationMark, "A loaded skinned mesh immediately counts as the spawned model");
        emptySkinned.sharedMesh = null;
        Expect(!npc.ShouldShowExclamationMark, "Removing the skinned mesh removes presence without destroying its component");

        foreach (bool skinned in new[] { false, true })
            for (int bits = 0; bits < 32; bits++)
            {
                npc.RenderedChildren.Clear();
                Renderer model = Model(skinned);
                model.enabled = (bits & 1) != 0;
                model.forceRenderingOff = (bits & 2) != 0;
                model.gameObject.activeInHierarchy = (bits & 4) != 0;
                model.isVisible = (bits & 8) != 0;
                bool labelOnly = (bits & 16) != 0;
                if (labelOnly) model.gameObject.transform.AddComponent(new TMPro.TMP_Text());
                npc.RenderedChildren.Add(model);
                Expect(npc.ShouldShowExclamationMark == (model.enabled && !model.forceRenderingOff &&
                    model.gameObject.activeInHierarchy && !labelOnly),
                    "Present model eligibility is independent of camera culling; model type " + skinned + ", bits " + bits);
            }

        npc.RenderedChildren.Clear();
        MeshRenderer body = (MeshRenderer)Model();
        body.isVisible = false;
        npc.RenderedChildren.Add(body);
        Expect(npc.ShouldShowExclamationMark, "An NPC offscreen on the gameplay camera still appears on the map");
        body.gameObject.activeInHierarchy = false;
        Expect(!npc.ShouldShowExclamationMark, "A hidden model parent hides its model even while progression holder remains active");
        body.gameObject.activeInHierarchy = true;
        Expect(npc.ShouldShowExclamationMark, "Restoring a model parent immediately restores presence eligibility");
        npc.gameObject.scene.isLoaded = false;
        Expect(!npc.ShouldShowExclamationMark, "No presence marker survives scene unload");
        npc.gameObject.scene.isLoaded = true;
        npc.RenderedChildren.Clear();
        Expect(!npc.ShouldShowExclamationMark, "Despawn removes a previously present model without needing a new phase");
        npc.RenderedChildren.Add(Model(true));
        Expect(npc.ShouldShowExclamationMark, "Respawn refreshes current children instead of trusting stale renderer discovery");
        npc.RenderedChildren.Add(nameLabel);
        Expect(npc.ShouldShowExclamationMark, "An auxiliary label does not invalidate an otherwise present body");
    }

    private static Renderer Model(bool skinned = false)
    {
        if (skinned) return new SkinnedMeshRenderer { sharedMesh = new Mesh() };
        MeshRenderer mesh = new MeshRenderer();
        mesh.gameObject.transform.AddComponent(new MeshFilter { sharedMesh = new Mesh() });
        return mesh;
    }

    private static void SpawnReadiness()
    {
        NPCProgressionManager npc = Create();
        npc.phaseHasArrived = true;
        Expect(!NPCSpawnConditionMonitor.CanShowNPCAttentionMarker(null), "Missing/destroyed NPC never has spawn permission");
        NPCSpawnerCondition gate = new NPCSpawnerCondition { npcRoot = npc.gameObject };
        Expect(!gate.CanShowAttentionMarkerFor(null), "Spawner refuses missing NPC identity");
        NPCSpawnConditionMonitor.Register(gate);
        Expect(!npc.ShouldShowExclamationMark, "Authored-active NPC before initial spawn evaluation does not flash a marker");
        gate.evaluated = true;
        gate.lastRuleVisibility = false;
        Expect(!npc.ShouldShowExclamationMark, "Waiting for manual SpawnNPC keeps the marker absent even if the model was authored active");
        gate.manualOverride = 1;
        Expect(npc.ShouldShowExclamationMark, "Manual spawn reveals a present model without waiting for automatic rules");
        gate.manualOverride = -1;
        Expect(!npc.ShouldShowExclamationMark, "Manual despawn vetoes the marker even before model hiding settles");
        gate.manualOverride = 0;
        gate.lastRuleVisibility = true;
        Expect(npc.ShouldShowExclamationMark, "A successfully evaluated visible spawn rule authorizes the arrived model");
        gate.lastRuleVisibility = false;
        Expect(!npc.ShouldShowExclamationMark, "Changing spawn rules to hidden revokes eligibility immediately");
        gate.enabled = false;
        Expect(npc.ShouldShowExclamationMark, "An intentionally disabled debug gate bypasses rules when the NPC is actually visible");
        gate.npcRoot.activeInHierarchy = false;
        Expect(!npc.ShouldShowExclamationMark, "Disabled rules never resurrect a genuinely hidden controlled object");
        gate.npcRoot.activeInHierarchy = true;

        foreach (int manual in new[] { -1, 0, 1 })
            for (int bits = 0; bits < 16; bits++)
            {
                gate.enabled = (bits & 1) != 0;
                gate.evaluated = (bits & 2) != 0;
                gate.lastRuleVisibility = (bits & 4) != 0;
                gate.npcRoot.activeInHierarchy = (bits & 8) != 0;
                gate.manualOverride = manual;
                bool expected = gate.npcRoot.activeInHierarchy && (!gate.enabled || manual > 0 ||
                    (manual == 0 && gate.evaluated && gate.lastRuleVisibility));
                Expect(gate.CanShowAttentionMarkerFor(npc.transform) == expected, "Spawn authorization state matrix, bits " + bits + ", override " + manual);
                Expect(npc.ShouldShowExclamationMark == expected, "Shared map/overhead property honors spawn authorization, bits " + bits + ", override " + manual);
            }
        npc.gameObject.activeInHierarchy = true;
        gate.enabled = true;
        gate.manualOverride = 1;
        NPCSpawnConditionMonitor.AddDestroyedEntry();
        Expect(npc.ShouldShowExclamationMark, "Destroyed registered spawn gates are harmless");

        NPCSpawnerCondition unrelated = new NPCSpawnerCondition();
        NPCSpawnConditionMonitor.Register(unrelated);
        Expect(npc.ShouldShowExclamationMark, "An unrelated unevaluated spawner cannot hide another NPC's markers");
        GameObject enclosingRoot = new GameObject();
        npc.transform.parent = enclosingRoot.transform;
        NPCSpawnerCondition enclosingGate = new NPCSpawnerCondition { npcRoot = enclosingRoot };
        NPCSpawnConditionMonitor.Register(enclosingGate);
        Expect(!npc.ShouldShowExclamationMark, "A controller of the containing NPC hierarchy vetoes pre-spawn markers");
        enclosingGate.evaluated = enclosingGate.lastRuleVisibility = true;
        Expect(npc.ShouldShowExclamationMark, "Every related containing gate must allow the NPC");
        GameObject bodyChild = new GameObject();
        bodyChild.transform.parent = npc.transform;
        NPCSpawnerCondition bodyGate = new NPCSpawnerCondition { npcRoot = bodyChild };
        NPCSpawnConditionMonitor.Register(bodyGate);
        Expect(!npc.ShouldShowExclamationMark, "A separate controller of the NPC's model subtree vetoes pre-spawn markers");
        bodyGate.manualOverride = 1;
        Expect(npc.ShouldShowExclamationMark, "Shown model subtree plus shown containing hierarchy authorize the NPC");
        bodyChild.activeInHierarchy = false;
        Expect(!npc.ShouldShowExclamationMark, "Manual Spawn cannot ignore a still-hidden model hierarchy");
        bodyChild.activeInHierarchy = true;
        NPCSpawnConditionMonitor.Unregister(bodyGate);
        Expect(npc.ShouldShowExclamationMark, "Removing a gate releases its old suppression without a domain reload");
        NPCSpawnConditionMonitor.Unregister(enclosingGate);
        NPCSpawnConditionMonitor.Unregister(gate);
        npc.transform.parent = null;
        NPCSpawnerCondition attachedGate = new NPCSpawnerCondition { gameObject = npc.gameObject };
        NPCSpawnConditionMonitor.Register(attachedGate);
        Expect(!npc.ShouldShowExclamationMark, "An attached spawner with no explicit npcRoot still protects its actual NPC");
        attachedGate.evaluated = attachedGate.lastRuleVisibility = true;
        Expect(npc.ShouldShowExclamationMark, "Attached spawned condition also supports implicit controlled object");
        NPCSpawnConditionMonitor.Reset();
    }

    private static void DestinationResolution()
    {
        NPCProgressionManager manager = Create();
        Expect(!manager.Reached(-1), "Invalid negative phase is not arrived");
        Expect(!manager.Reached(1), "Out of range phase is not arrived");
        manager.transform.position = new Vector3(10f, 2f, 20f);
        Expect(manager.Reached(0), "Exact destination is arrived");
        manager.transform.position = new Vector3(10.24f, 2f, 20f);
        Expect(manager.Reached(0), "Minimum .25m radius allows grounded placement");
        manager.transform.position = new Vector3(10.26f, 2f, 20f);
        Expect(!manager.Reached(0), "Unarrived nearby NPC is excluded");
        manager.transform.position = new Vector3(10f, 3.49f, 20f);
        Expect(manager.Reached(0), "Ground projection within 1.5m is allowed");
        manager.transform.position = new Vector3(10f, 3.51f, 20f);
        Expect(!manager.Reached(0), "Wrong-floor marker is excluded");
        manager.arrivalPadding = .3f;
        manager.waypointArrivalDistance = .2f;
        manager.transform.position = new Vector3(10.49f, 2f, 20f);
        Expect(manager.Reached(0), "Configured route tolerance is honored");
        manager.transform.position = new Vector3(10.51f, 2f, 20f);
        Expect(!manager.Reached(0), "Configured route tolerance remains bounded");

        manager = Create();
        manager.authoredStartPosition = new Vector3(3f, 1f, 4f);
        manager.phases[0].targetLocation = null;
        manager.transform.position = manager.authoredStartPosition;
        Expect(manager.Reached(0), "Targetless first phase uses authored spawn");
        manager.transform.position = Vector3.zero;
        Expect(!manager.Reached(0), "Targetless first phase is not anywhere in world");
        manager.phases[0].targetLocation = manager.transform;
        Expect(!manager.Reached(0), "Self marker cannot follow a displaced NPC and claim arrival");
        manager.transform.position = manager.authoredStartPosition;
        Expect(manager.Reached(0), "Self marker returns to authored spawn");

        manager = Create();
        manager.authoredStartPosition = Vector3.zero;
        manager.phases.Add(new NPCProgressionPhase { showExclamationMark = true });
        manager.phases.Add(new NPCProgressionPhase { showExclamationMark = true });
        manager.transform.position = new Vector3(10f, 2f, 20f);
        Expect(manager.Reached(2), "Targetless later phase inherits last authored destination");
        manager.transform.position = manager.authoredStartPosition;
        Expect(!manager.Reached(2), "Later targetless phase does not return to initial spawn");
        manager.phases[0].targetLocation = manager.transform;
        Expect(manager.Reached(2), "Inherited legacy self marker resolves authored start");
        manager.phases[0] = null;
        Expect(manager.Reached(2), "Null earlier phase safely falls through to authored spawn");
        manager.phases[2] = null;
        Expect(!manager.Reached(2), "Missing active phase never arrived");

        manager = Create();
        manager.movementMode = NPCProgressionMovementMode.Waypoints;
        manager.phases[0].targetLocation.position = new Vector3(10f, 12f, 20f);
        manager.transform.position = new Vector3(10f, 2f, 20f);
        manager.GroundProbeSucceeds = true;
        manager.GroundProbeHeight = 2f;
        Expect(manager.Reached(0), "Waypoint arrival compares actual ground below elevated authored marker");
        Expect(manager.GroundProbeCount == 1, "Waypoint destination uses the ground-projection boundary");
        manager.GroundProbeSucceeds = false;
        Expect(!manager.Reached(0), "Missing ground falls back to authored height rather than inventing arrival");
        manager.transform.position = manager.phases[0].targetLocation.position;
        Expect(manager.Reached(0), "Missing ground preserves authored placement semantics");
        manager.GroundProbeSucceeds = true;
        manager.movementMode = NPCProgressionMovementMode.NavMesh;
        manager.GroundProbeCount = 0;
        manager.transform.position = new Vector3(10f, 2f, 20f);
        Expect(!manager.Reached(0), "NavMesh arrival does not substitute waypoint grounding");
        Expect(manager.GroundProbeCount == 0, "NavMesh eligibility avoids unnecessary terrain projection");
    }

    private static void ArrivalLifecycle()
    {
        NPCProgressionManager manager = Create();
        manager.transform.position = Vector3.zero;
        manager.Activate(0);
        Expect(!manager.ShouldShowExclamationMark, "Activation while not placed is not arrival");
        manager.transform.position = manager.phases[0].targetLocation.position;
        manager.Activate(0);
        Expect(manager.ShouldShowExclamationMark, "Placed active phase exposes marker");
        manager.isActiveAndEnabled = false;
        Expect(!manager.ShouldShowExclamationMark, "Disabled arrived NPC hidden");
        manager.isActiveAndEnabled = true;
        Expect(manager.ShouldShowExclamationMark, "Re-enabled arrived NPC restores marker");
        manager.BeginTravel();
        Expect(!manager.ShouldShowExclamationMark, "Travel hides departed phase immediately");
        manager.movementRoutine = null;
        Expect(!manager.ShouldShowExclamationMark, "Interrupted travel remains unarrived even after coroutine stops");
        manager.FinishTravel(0);
        Expect(manager.ShouldShowExclamationMark, "Arrival restores marker");
        manager.phases.Add(new NPCProgressionPhase { showExclamationMark = false, targetLocation = new Transform { position = new Vector3(30f, 2f, 20f) } });
        manager.BeginTravel();
        manager.transform.position = manager.phases[1].targetLocation.position;
        manager.FinishTravel(1);
        Expect(!manager.ShouldShowExclamationMark, "Arrived unchecked phase is hidden");
        manager.phases[1].showExclamationMark = true;
        Expect(manager.ShouldShowExclamationMark, "Current phase checkbox can be enabled live");
        FusionConnectionManager.Instance = new FusionConnectionManager { IsGuestInHostWorld = true };
        Expect(!manager.ShouldShowExclamationMark, "Visiting guest cannot see own progression cue");
    }

    private static void GraphicGeometry()
    {
        NPCExclamationGraphic graphic = new NPCExclamationGraphic();
        Expect(!graphic.raycastTarget, "Badge never becomes a clickable obstacle");
        foreach (float size in new[] { 16f, 22f, 28f, 32f, 42f, 56f, 112f })
        {
            VertexHelper mesh = graphic.Render(new Rect(-size * .5f, -size * .5f, size, size));
            Expect(mesh.Positions.Count == 94, "Simple bounded vertex count at size " + size);
            Expect(mesh.Indices.Count == 258, "Stable triangle count at size " + size);
            bool sawGold = false, sawInk = false;
            foreach (int index in mesh.Indices)
                Expect(index >= 0 && index < mesh.Positions.Count, "All triangles reference real vertices");
            for (int i = 0; i < mesh.Positions.Count; i++)
            {
                Vector2 position = mesh.Positions[i];
                Expect(!float.IsNaN(position.x) && !float.IsNaN(position.y), "No invalid mesh point");
                Expect(Math.Abs(position.x) <= size * .5f + .001f && Math.Abs(position.y) <= size * .5f + .001f,
                    "Mesh fits declared badge area");
                Color32 color = mesh.Colors[i];
                Expect(color.a == 255, "Every badge vertex is opaque");
                sawGold |= color.r == 255 && color.g == 198 && color.b == 55;
                sawInk |= color.r == 65 && color.g == 40 && color.b == 25;
            }
            Expect(sawGold && sawInk, "Contrasting ring/field/glyph colors present");
            if (size == 22f || size == 28f || size == 32f)
            {
                // Actual vector mesh: two 30-vertex discs precede the four-vertex bar,
                // then the 30-vertex dot. These minimum physical-pixel dimensions
                // keep the reduced marker distinguishable without a font glyph.
                float barWidth = mesh.Positions[62].x - mesh.Positions[60].x;
                float barHeight = mesh.Positions[61].y - mesh.Positions[60].y;
                float dotDiameter = 2f * (mesh.Positions[65].x - mesh.Positions[64].x);
                float rim = mesh.Positions[1].x - mesh.Positions[31].x;
                Expect(barWidth >= 3f && barHeight >= 7.5f, "Reduced badge retains a substantial exclamation bar");
                Expect(dotDiameter >= 3f, "Reduced badge retains a separated readable dot");
                Expect(rim >= 1.7f, "Reduced badge retains a high-contrast outer outline");
            }
        }
        Expect(graphic.Render(new Rect(0f, 0f, 0f, 42f)).Positions.Count == 0,
            "Zero-area badge has no stale mesh");
    }

    private static void MapProjectionAndSize()
    {
        Vector3 one = new Vector3(1f, 1f, 1f);
        foreach (float pixels in new[] { 16f, 22f, 28f, 32f, 42f, 56f })
            foreach (float canvasScale in new[] { .3f, .5f, 1f, 1.5f, 2f, 3f })
                foreach (float relative in new[] { .2f, .5f, 1f, 1.4f, 2f })
                {
                    Vector2 size = MapProjection.Size(pixels, canvasScale,
                        new Vector3(relative, relative, 1f), one);
                    Expect(Math.Abs(size.x * canvasScale * relative - pixels) < .001f,
                        "NPC map badge stays constant screen pixels horizontally");
                    Expect(Math.Abs(size.y * canvasScale * relative - pixels) < .001f,
                        "NPC map badge stays constant screen pixels vertically");
                }
        Vector2 nonuniform = MapProjection.Size(42f, 2f, new Vector3(2f, .5f, 1f), one);
        Expect(Math.Abs(nonuniform.x - 10.5f) < .001f && Math.Abs(nonuniform.y - 42f) < .001f,
            "Nonuniform map scaling preserves round screen badge");
        Vector2 safe = MapProjection.Size(float.NaN, float.PositiveInfinity,
            new Vector3(float.NaN, float.NegativeInfinity, 1f), one);
        Expect(safe.x == 22f && safe.y == 22f, "Malformed sizing inputs safely fall back to compact size");

        Rect area = new Rect(0f, 0f, 200f, 100f), uv = new Rect(0f, 0f, 1f, 1f);
        Vector2 point;
        Expect(MapProjection.Project(new Vector3(.5f, .5f, 1f), area, uv, out point) && point.x == 0f && point.y == 0f,
            "NPC at map center projects to local center");
        Expect(MapProjection.Project(new Vector3(.25f, .75f, 1f), area, uv, out point) && point.x == -50f && point.y == 25f,
            "NPC marker matches actual map location");
        foreach (Vector3 outside in new[] { new Vector3(-.01f, .5f, 1f), new Vector3(1.01f, .5f, 1f),
            new Vector3(.5f, -.01f, 1f), new Vector3(.5f, 1.01f, 1f), new Vector3(.5f, .5f, 0f),
            new Vector3(.5f, .5f, -1f), new Vector3(float.NaN, .5f, 1f), new Vector3(.5f, float.PositiveInfinity, 1f) })
            Expect(!MapProjection.Project(outside, area, uv, out point), "Off-map/invalid NPC must never be clamped to a false edge location");
        Expect(MapProjection.Project(new Vector3(.25f, .75f, 1f), area, new Rect(.25f, .25f, .5f, .5f), out point) &&
            point.x == -100f && point.y == 50f, "Cropped map texture UV matches marker position");
        Expect(!MapProjection.Project(new Vector3(.1f, .5f, 1f), area, new Rect(.25f, .25f, .5f, .5f), out point),
            "UV crop excludes NPC outside visible image");
        Expect(!MapProjection.Project(new Vector3(.5f, .5f, 1f), new Rect(0f, 0f, 0f, 100f), uv, out point),
            "Invalid image area cannot expose misplaced marker");
        Expect(!MapProjection.Project(new Vector3(.5f, .5f, 1f), area, new Rect(0f, 0f, 0f, 1f), out point),
            "Invalid UV extent cannot expose misplaced marker");
    }

    private static RectTransform VisibleHUD(Rect bounds)
    {
        RectTransform rect = new RectTransform { ScreenBounds = bounds };
        rect.AddComponent(new Canvas());
        return rect;
    }

    private static void HUDReservationGeometry()
    {
        UIReservedRegionLayout.Reset();
        Rect badge = new Rect(90f, 90f, 32f, 32f);
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(badge), "Unregistered HUD never hides an overhead marker");
        UIReservedRegionLayout.Register(null);
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(badge), "Null registration cannot hide a marker");
        RectTransform map = VisibleHUD(new Rect(100f, 100f, 200f, 150f));
        UIReservedRegionLayout.Register(map);
        UIReservedRegionLayout.Register(map);
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(badge), "Corner intersection hides even when projected center is outside the map");
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(284f, 130f, 32f, 32f)), "The full right edge, not only the center, is protected");
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(130f, 234f, 32f, 32f)), "The full upper edge is protected");
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(140f, 140f, 32f, 32f)), "Badge entirely inside map is suppressed");
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(60f, 120f, 32f, 32f)), "Unrelated world-space area stays visible");
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(68f, 120f, 32f, 32f)), "Touching edge without padding has no pixel overlap");
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(68f, 120f, 32f, 32f), 2f), "Two-pixel safety padding protects a touching rim");
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(66f, 120f, 32f, 32f), 2f), "Exactly separated safety margin is permitted");
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(66.1f, 120f, 32f, 32f), 2f), "A fractional-pixel safety-margin crossing is suppressed");
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(66.1f, 120f, 32f, 32f), -2f), "Negative padding cannot shrink protected HUD bounds");
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(150f, 150f, 0f, 32f)), "Zero-width candidate is safely ignored");
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(150f, 150f, 32f, -1f)), "Negative-height candidate is safely ignored");
        map.ScreenBounds = new Rect(600f, 400f, 200f, 150f);
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(badge), "Moving or resizing minimap releases its previous position without stale caching");
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(590f, 390f, 32f, 32f)), "Current minimap position is protected in the same frame");
        map.BoundsValid = false;
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(590f, 390f, 32f, 32f)), "Failed engine projection cannot invent reserved bounds");
        map.BoundsValid = true;
        UIReservedRegionLayout.Unregister(map);
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(new Rect(590f, 390f, 32f, 32f)), "One unregistration removes a repeatedly registered HUD");

        Random random = new Random(310514);
        foreach (Vector2 screen in new[] { new Vector2(1280f, 720f), new Vector2(1024f, 768f), new Vector2(1920f, 1080f), new Vector2(720f, 1280f) })
        {
            UIReservedRegionLayout.Reset();
            Rect mapBounds = new Rect(screen.x - 230f, screen.y - 200f, 210f, 180f);
            map = VisibleHUD(mapBounds);
            UIReservedRegionLayout.Register(map);
            for (int index = 0; index < 250; index++)
            {
                float centerX = mapBounds.xMin - 50f + (float)random.NextDouble() * (mapBounds.width + 100f);
                float centerY = mapBounds.yMin - 50f + (float)random.NextDouble() * (mapBounds.height + 100f);
                Rect candidate = new Rect(centerX - 16f, centerY - 16f, 32f, 32f);
                bool expected = centerX + 16f > mapBounds.xMin - 2f && centerX - 16f < mapBounds.xMax + 2f &&
                    centerY + 16f > mapBounds.yMin - 2f && centerY - 16f < mapBounds.yMax + 2f;
                Expect(UIReservedRegionLayout.OverlapsVisibleReservation(candidate, 2f) == expected,
                    "Physical-pixel minimap exclusion agrees with full-badge oracle on " + screen.x + "x" + screen.y);
            }
        }
        UIReservedRegionLayout.Reset();
    }

    private static void HUDReservationVisibility()
    {
        UIReservedRegionLayout.Reset();
        Rect candidate = new Rect(120f, 120f, 32f, 32f);
        RectTransform map = VisibleHUD(new Rect(100f, 100f, 200f, 150f));
        UIReservedRegionLayout.Register(map);
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Visible minimap owns its screen area");
        map.gameObject.activeInHierarchy = false;
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Hidden/locked minimap does not conceal the NPC badge");
        map.gameObject.activeInHierarchy = true;
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Showing map restores its screen reservation immediately");
        map.TryGetComponent(out Canvas canvas);
        canvas.isActiveAndEnabled = false;
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Disabled minimap Canvas does not reserve invisible screen space");
        canvas.isActiveAndEnabled = true;
        CanvasGroup mapGroup = new CanvasGroup { alpha = 0f };
        map.AddComponent(mapGroup);
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Fully faded minimap does not hide the overhead badge");
        mapGroup.alpha = .002f;
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "A still-rendered minimap retains exclusion");
        mapGroup.alpha = .001f;
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Opacity at visibility threshold is excluded");
        mapGroup.alpha = 1f;
        Transform outer = new Transform();
        CanvasGroup parentGroup = new CanvasGroup { alpha = 0f };
        outer.AddComponent(parentGroup);
        map.parent = outer;
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Invisible ancestor group hides the minimap reservation");
        mapGroup.ignoreParentGroups = true;
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Ignore-parent opacity keeps a genuinely visible map protected");
        Canvas disabledParentCanvas = new Canvas { isActiveAndEnabled = false };
        outer.AddComponent(disabledParentCanvas);
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Ignoring parent groups cannot override a disabled parent Canvas");
        disabledParentCanvas.isActiveAndEnabled = true;
        mapGroup.ignoreParentGroups = false;
        parentGroup.alpha = .01f;
        mapGroup.alpha = .01f;
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Combined ancestor opacity is evaluated, not each group alone");
        UIReservedRegionLayout.Unregister(map);
        RectTransform orphan = new RectTransform { ScreenBounds = map.ScreenBounds };
        UIReservedRegionLayout.Register(orphan);
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Unrendered orphan rectangles never conceal NPC markers");
        RectTransform secondMap = VisibleHUD(map.ScreenBounds);
        UIReservedRegionLayout.Register(secondMap);
        Expect(UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "One visible reservation still protects HUD when another is invalid");
        UIReservedRegionLayout.Unregister(secondMap);
        Expect(!UIReservedRegionLayout.OverlapsVisibleReservation(candidate), "Teardown releases the visible map without needing domain reload");
        UIReservedRegionLayout.Reset();
    }

    private static void Expect(bool condition, string message)
    {
        assertions++;
        if (!condition) throw new InvalidOperationException(message);
    }
