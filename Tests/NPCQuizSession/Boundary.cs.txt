namespace UnityEngine.SceneManagement
{
    public struct Scene
    {
        public static HashSet<int> unloaded = new HashSet<int>();
        public int id;
        public Scene(int id) { this.id = id; }
        public bool IsValid() => id > 0;
        public bool isLoaded => IsValid() && !unloaded.Contains(id);
        public static bool operator ==(Scene a, Scene b) => a.id == b.id;
        public static bool operator !=(Scene a, Scene b) => a.id != b.id;
        public override bool Equals(object other) => other is Scene scene && scene.id == id;
        public override int GetHashCode() => id;
    }
}

public sealed class PlayerDataManager
{
    public static PlayerDataManager Instance;
    public PlayerData CurrentData = new PlayerData();
    public bool saveSucceeds = true;
    public int saves;
    public Action saveHook;
    public List<QuizProgressSaveData> committed = new List<QuizProgressSaveData>();
    public bool TrySaveGame()
    {
        saves++;
        saveHook?.Invoke();
        if (!saveSucceeds) return false;
        committed = new List<QuizProgressSaveData>();
        foreach (QuizProgressSaveData record in CurrentData.quizProgress) committed.Add(record.Clone());
        return true;
    }

public sealed class QuizView : Behaviour
{
    public GameObject Panel = new GameObject();
    public bool IsReady = true;
    public int shownQuestion = -1, completionShows, feedbackShows, saveErrors, selection = -1;
    public bool busy, correct;
    public string feedback, error;
    public QuizUILayout layout = new QuizUILayout();
    public bool npcPresentation;
    public bool HasNpcPresentation => npcPresentation;
    public string npcName;
    public void SetNpcPresentation(bool active, string speakerName) { npcPresentation = active; npcName = speakerName; QuizNPCPresentation.Trace.Add("view:" + active); }
    public event Action<int> ChoiceSelected;
    public event Action SubmitRequested, NextRequested, PreviousRequested, FinishRequested, LeaveRequested, RetrySaveRequested, PauseRequested;
    public int score = -1;
    public void ShowAssessmentQuestion(QuizData data, int index, int selected, int answeredCount)
    { ShowQuestion(data, index, null); selection = selected; }
    public IReadOnlyDictionary<string, int> resultAnswers;
    public void ShowAssessmentResult(QuizData data, int value, IReadOnlyDictionary<string, int> answers = null)
    {
        ShowCompletion(data); score = value;
        resultAnswers = answers == null ? null : new Dictionary<string, int>(answers);
    }
    public void Reset() { selection = shownQuestion = -1; completionShows = feedbackShows = saveErrors = 0; resultAnswers = null; }
    public void SetSelection(int index) { selection = index; }
    public void ShowQuestion(QuizData data, int index, IReadOnlyCollection<string> ids)
    {
        if (index < 0 || index >= data.QuestionCount) throw new InvalidOperationException("Controller passed invalid question index to view.");
        shownQuestion = index; selection = -1;
    }
    public void ShowCompletion(QuizData data) { shownQuestion = -1; completionShows++; }
    public void ShowFeedback(bool correct, string text) { this.correct = correct; feedback = text; feedbackShows++; }
    public void ShowSaveError(string text) { error = text; saveErrors++; }
    public void SetBusy(bool value) { busy = value; }
    public void Choose(int index) { ChoiceSelected?.Invoke(index); }
    public void Submit() { SubmitRequested?.Invoke(); }
    public void Next() { NextRequested?.Invoke(); }
    public void Previous() { PreviousRequested?.Invoke(); }
    public void Finish() { FinishRequested?.Invoke(); }
    public void Leave() { LeaveRequested?.Invoke(); }
    public void Retry() { RetrySaveRequested?.Invoke(); }
    public void Pause() { PauseRequested?.Invoke(); }
}
public sealed class UIPanelCoordinator
{
    public static UIPanelCoordinator Instance;
    private readonly List<GameObject> panels = new List<GameObject>();
    public bool HasOpenPanel => panels.Count > 0;
    public void OpenPanel(GameObject panel)
    {
        QuizNPCPresentation.Trace.Add("panel:open");
        if (panels.Count > 0) panels[panels.Count - 1].SetActive(false);
        panels.Remove(panel); panels.Add(panel); panel.SetActive(true);
    }
    public void ClosePanel(GameObject panel)
    {
        QuizNPCPresentation.Trace.Add("panel:close");
        if (panel != null) panel.SetActive(false);
        panels.Remove(panel); if (panels.Count > 0) panels[panels.Count - 1].SetActive(true);
    }
    public bool IsOpen(GameObject panel) => panels.Contains(panel);
    public bool ContainsPanel(GameObject panel) => panels.Contains(panel);
    public bool IsTopPanel(GameObject panel) => panels.Count > 0 && panels[panels.Count - 1] == panel;
}

// Camera/animation boundary: the production controller owns this double exactly
// like its scene-local presenter. Camera mathematics is verified separately.
public sealed class QuizNPCPresentation : MonoBehaviour
{
    public static readonly List<string> Trace = new List<string>();
    public static bool BeginSucceeds = true;
    public static readonly List<QuizNPCPresentation> Created = new List<QuizNPCPresentation>();
    public bool Active;
    public bool IsActive => Active;
    public string SpeakerName => "Bhan";
    public UnityEngine.Object receivedOwner;
    public Rect viewport;
    public int beginCalls, viewportCalls, questionCalls, endCalls;
    public bool speaking, restored;
    private InputManager localInput;
    private bool lookComponentBefore;
    public QuizNPCPresentation() { Created.Add(this); }
    public static void ResetBoundary() { Trace.Clear(); Created.Clear(); BeginSucceeds = true; }
    public bool Begin(UnityEngine.Object owner, Rect region)
    {
        receivedOwner = owner; viewport = region; beginCalls++;
        Trace.Add("presenter:begin"); Active = BeginSucceeds;
        if (Active)
        {
            localInput = UnityEngine.Object.FindObjectOfType<InputManager>();
            if (localInput != null) { lookComponentBefore = localInput.lookComponentEnabled; localInput.lookComponentEnabled = false; }
        }
        return Active;
    }
    public void SetViewport(Rect region) { viewport = region; viewportCalls++; Trace.Add("presenter:viewport"); }
    public void PresentQuestion() { questionCalls++; speaking = true; Trace.Add("presenter:question"); }
    public void SetSpeaking(bool value) { speaking = value; Trace.Add("presenter:speaking:" + value); }
    public void End(bool restoreCamera)
    {
        if (!Active) return;
        if (localInput != null) localInput.lookComponentEnabled = lookComponentBefore;
        localInput = null;
        endCalls++; restored = restoreCamera; speaking = Active = false; Trace.Add("presenter:end");
    }
}
