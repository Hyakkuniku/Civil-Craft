public sealed class UIPanelCoordinator
{
    private sealed class ObjectState { public GameObject target; public bool wasActive; }
    private sealed class CanvasState { public Canvas target; public bool wasEnabled; }
    private sealed class PanelFrame
    {
        public GameObject panel;
        public readonly List<ObjectState> previousStates = new List<ObjectState>();
        public readonly List<CanvasState> previousCanvasStates = new List<CanvasState>();
    }
    public static UIPanelCoordinator Instance;
    private readonly Stack<PanelFrame> panelStack = new Stack<PanelFrame>();
    public int Frames => panelStack.Count;
    public void ClosePanel(GameObject panel) { ClosePanel(panel, true); }
    // Opening/capturing HUD engine objects is an explicit test setup boundary.
    // Closing/unwinding and restoring the captured states use production code.
    public void CaptureAndOpen(GameObject panel, GameObject hud, Canvas canvas)
    {
        var frame = new PanelFrame { panel = panel };
        frame.previousStates.Add(new ObjectState { target = hud, wasActive = hud.activeSelf });
        frame.previousCanvasStates.Add(new CanvasState { target = canvas, wasEnabled = canvas.enabled });
        hud.SetActive(false); canvas.enabled = false; panel.SetActive(true); panelStack.Push(frame);
    }
    // MODAL_RESTORE
}

public sealed class LessonUIManager

    public LessonData CurrentLesson;
    public bool IsOpen => lessonPanel != null && lessonPanel.activeSelf;
    public event Action<LessonData> LessonClosed;
    public object resetScrollRoutine;
    public int ScrollStops, ControlRestores;
    private void StopCoroutine(object routine) { ScrollStops++; }
    private void RestorePlayerControls() { ControlRestores++; }
    public void CaptureCanvas(Canvas canvas) { hiddenCanvasStates.Add(new CanvasState { canvas = canvas, wasEnabled = canvas.enabled }); canvas.enabled = false; }
    // LESSON_CLOSE
}
