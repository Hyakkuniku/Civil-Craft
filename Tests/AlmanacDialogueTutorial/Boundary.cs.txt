    public class Coroutine { public IEnumerator routine; public bool stopped; }
    public class MonoBehaviour
    {
        public bool isActiveAndEnabled=true;
        public readonly List<Coroutine> coroutines=new List<Coroutine>();
        public Coroutine StartCoroutine(IEnumerator routine)
        {
            var task=new Coroutine{routine=routine}; coroutines.Add(task);
            if(!routine.MoveNext()) task.stopped=true;
            return task;
        }
        public void StopCoroutine(Coroutine task) { if(task!=null)task.stopped=true; }
        public void Tick()
        {
            foreach(var task in coroutines.ToArray())
                if(!task.stopped&&!task.routine.MoveNext())task.stopped=true;
            coroutines.RemoveAll(task=>task.stopped);
        }
        public int Running=>coroutines.Count(task=>!task.stopped);
    }

public class TutorialSequence
{
    public string lessonName;
    public TutorialStep[] tutorialSteps={new TutorialStep(),new TutorialStep(),new TutorialStep()};
    public bool eligible=true;
    public bool CanStartAsPriorityTutorial()=>eligible&&!(PlayerDataManager.Instance?.CurrentData?.completedLessons?.Contains(lessonName)??false);
    public void TryStartTutorial()=>TutorialManager.Instance?.PlayPriorityTutorial(this);
}
public class TutorialManager
{
    public static TutorialManager Instance;
    public TutorialSequence ActiveSequence; public int CurrentStepIndex,Starts,Cancels,Completes,Holds,Releases,Resumed,Advances;
    public bool IsTutorialActive=>ActiveSequence!=null;
    public event Action<TutorialSequence> OnTutorialCompleted;
    public readonly Stack<TutorialSequence> suspended=new Stack<TutorialSequence>();
    public bool IsPlayingSequence(TutorialSequence sequence)=>sequence!=null&&ReferenceEquals(ActiveSequence,sequence);
    public void PlayPriorityTutorial(TutorialSequence sequence)
    {
        if(sequence==null||!sequence.CanStartAsPriorityTutorial()||IsPlayingSequence(sequence))return;
        if(ActiveSequence!=null)suspended.Push(ActiveSequence);
        ActiveSequence=sequence;CurrentStepIndex=0;Starts++;
    }
    public void BeginSuspendedTutorialResumeHold(){Holds++;}
    public void EndSuspendedTutorialResumeHold(){if(Holds>0)Holds--;Releases++;ResumeIfFree();}
    void ResumeIfFree(){if(Holds==0&&ActiveSequence==null&&suspended.Count>0){ActiveSequence=suspended.Pop();Resumed++;}}
    public bool CancelTutorialWithoutCompletion(TutorialSequence sequence)
    {
        if(!IsPlayingSequence(sequence))return false;
        ActiveSequence=null;Cancels++;ResumeIfFree();return true;
    }
    public void ShowNextStep(){CurrentStepIndex++;Advances++;}
    public void Finish()
    {
        var done=ActiveSequence;if(done==null)return;
        ActiveSequence=null;Completes++;
        PlayerDataManager.Instance.CurrentData.completedLessons.Add(done.lessonName);
        OnTutorialCompleted?.Invoke(done);ResumeIfFree();
    }
}
