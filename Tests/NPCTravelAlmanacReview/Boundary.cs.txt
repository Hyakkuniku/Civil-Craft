        public void StopCoroutine(Coroutine value){value.Stopped=true;}
        public void Tick(){foreach(var value in running.ToArray())if(!value.Stopped&&!value.Routine.MoveNext())value.Stopped=true;running.RemoveAll(value=>value.Stopped);}

public class TutorialManager
{
    public static TutorialManager Instance;public TutorialSequence Active;public List<TutorialSequence> Queued=new();public int Starts,Completes,CurrentStepIndex,Cancels;
    public void StartOrQueue(TutorialSequence sequence){Starts++;if(Active==null){Active=sequence;CurrentStepIndex=0;}else Queued.Add(sequence);}
    public bool IsPlayingSequence(TutorialSequence sequence)=>sequence!=null&&ReferenceEquals(Active,sequence);
    public void ShowNextStep()=>CurrentStepIndex++;
    public void CompleteTutorialIfPlaying(string id){if(Active==null||Active.lessonName!=id)return;PlayerDataManager.Instance.CurrentData.completedLessons.Add(id);Completes++;Active=null;}
    public void CancelTutorialWithoutCompletion(TutorialSequence sequence){if(ReferenceEquals(Active,sequence)){Cancels++;Active=null;}Queued.Remove(sequence);}
}
